cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico CACHE STRING "Fixed RP2040 board")
set(PICO_PLATFORM rp2040 CACHE STRING "Fixed processor")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "UF2 is generated by the workbench")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(workbench_firmware C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
pico_sdk_init()
add_executable(workbench main.c)
target_link_libraries(workbench pico_stdlib hardware_uart hardware_gpio)
pico_enable_stdio_uart(workbench 1)
pico_enable_stdio_usb(workbench 1)
pico_add_extra_outputs(workbench)
